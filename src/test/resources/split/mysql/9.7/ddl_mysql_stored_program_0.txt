CREATE PROCEDURE proc_stacked_diag()
BEGIN
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @stack_state = RETURNED_SQLSTATE, @stack_message = MESSAGE_TEXT;
    RESIGNAL;
  END;
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'stacked diagnostics';
END;
DROP EVENT IF EXISTS split97.ev_once;
DROP EVENT IF EXISTS split97.ev_once_renamed;
DROP EVENT IF EXISTS split97.ev_repeat;
CREATE DEFINER = CURRENT_USER EVENT IF NOT EXISTS split97.ev_once
  ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 1 HOUR + INTERVAL 1 MINUTE
  ON COMPLETION PRESERVE
  DISABLE
  COMMENT 'one time event'
  DO UPDATE split97.event_log SET note = 'once' WHERE id = 1;
CREATE EVENT split97.ev_repeat
  ON SCHEDULE EVERY 1 DAY
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 2 DAY
  ON COMPLETION NOT PRESERVE
  DISABLE ON SLAVE
  COMMENT 'repeat event'
  DO INSERT INTO split97.event_log VALUES (2, 'repeat');
ALTER EVENT split97.ev_once
  ON SCHEDULE EVERY 2 HOUR
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 1 DAY
  RENAME TO split97.ev_once_renamed
  ENABLE
  COMMENT 'renamed event'
  DO UPDATE split97.event_log SET note = 'renamed' WHERE id = 1;
ALTER EVENT split97.ev_repeat DISABLE ON SLAVE;
DROP EVENT IF EXISTS split97.ev_once_renamed;
DROP EVENT split97.ev_repeat;

CREATE PROCEDURE split_window_bugs.p_window_counts() BEGIN SELECT COUNT(*) OVER (ROWS BETWEEN 5 PRECEDING AND 5 FOLLOWING) AS a,COUNT(*) OVER (ROWS BETWEEN 5 PRECEDING AND 5 FOLLOWING)+1 AS a_plus,COUNT(*) OVER (ROWS BETWEEN 5 PRECEDING AND 5 FOLLOWING)-1 AS a_minus,COUNT(*) OVER () AS b,COUNT(*) OVER ()+1 AS b_plus,COUNT(*) OVER ()-1 AS b_minus FROM split_window_bugs.t_proc; END;











CREATE PROCEDURE p11() BEGIN DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; SELECT 1; GET CURRENT DIAGNOSTICS @n=NUMBER; GET STACKED DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; END; DROP TABLE missing_t; END;
CREATE PROCEDURE p12() BEGIN DECLARE CONTINUE HANDLER FOR SQLWARNING BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; RESIGNAL SET MYSQL_ERRNO=9999,MESSAGE_TEXT='mapped'; GET CURRENT DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m4=MESSAGE_TEXT; END; SIGNAL SQLSTATE '01001'; END;


CREATE PROCEDURE p15() BEGIN DECLARE done BOOL DEFAULT FALSE; DECLARE n INT; DECLARE c CURSOR FOR WITH RECURSIVE seq(n) AS (SELECT 1 UNION ALL SELECT n+1 FROM seq WHERE n<3) SELECT n FROM seq; DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=TRUE; OPEN c; FETCH c INTO n; CLOSE c; END;

  OUT output_value VECTOR(3),
  INOUT accumulator VECTOR(3)
)
