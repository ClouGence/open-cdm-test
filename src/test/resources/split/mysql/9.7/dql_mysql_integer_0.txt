WITH integer_cte AS (
  SELECT id,tiny_value,small_value,medium_value,int_value,big_value,signed_total
  FROM integer_generated
  WHERE signed_total BETWEEN -100 AND 100
)
SELECT id,signed_total
FROM integer_cte
WHERE big_value >= (
  SELECT AVG(big_value) FROM integer_cte
)
ORDER BY signed_total,id;
SELECT id,tiny_value,small_value,medium_value,int_value,big_value,
       SUM(big_value) OVER (
         PARTITION BY tiny_value
         ORDER BY id
         ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
       ) AS running_big,
       BIT_OR(int_value) OVER (ORDER BY id) AS running_mask
FROM integer_generated;
WITH RECURSIVE integer_sequence(n,tiny_value,small_value,medium_value,int_value,big_value) AS (
  SELECT 1,CAST(1 AS SIGNED),CAST(2 AS SIGNED),CAST(3 AS SIGNED),CAST(4 AS SIGNED),CAST(5 AS SIGNED)
  UNION ALL
  SELECT n+1,tiny_value+1,small_value+2,medium_value+3,int_value+4,big_value+5
  FROM integer_sequence
  WHERE n<4
)
SELECT n,tiny_value,small_value,medium_value,int_value,big_value
FROM integer_sequence;
