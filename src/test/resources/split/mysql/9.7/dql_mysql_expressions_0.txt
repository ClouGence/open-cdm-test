WITH src AS (SELECT 1 id,1 grp,3 v UNION ALL SELECT 2,1,5 UNION ALL SELECT 3,2,2) SELECT grp,SUM(v*2+id) total FROM src WHERE NOT v=2 OR id&1=0 GROUP BY grp HAVING SUM(v)+COUNT(*)*2>5 XOR grp=2 ORDER BY total>>1 DESC;
WITH src AS (SELECT 1 id,1 grp,3 v UNION ALL SELECT 2,1,5 UNION ALL SELECT 3,2,2) SELECT id,grp,v,SUM(v*2+id) OVER (PARTITION BY grp ORDER BY id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) running,RANK() OVER (ORDER BY v*2+id DESC) rnk FROM src WHERE NOT v BETWEEN 4 AND 6 XOR grp=2 ORDER BY running+rnk*2 DESC;
SELECT * FROM (SELECT id,grp,v,ROW_NUMBER() OVER (PARTITION BY grp ORDER BY v*2+id DESC) rn FROM (SELECT 1 id,1 grp,3 v UNION ALL SELECT 2,1,NULL UNION ALL SELECT 3,2,2) s) d WHERE rn+1*2<=3 AND NOT v IS NULL ORDER BY rn,v DESC;
WITH src AS (SELECT 1 id,JSON_OBJECT('n',2,'tags',JSON_ARRAY(1,3),'s','abc123') doc UNION ALL SELECT 2,JSON_OBJECT('n',4,'tags',JSON_ARRAY(4,5),'s','zzz')) SELECT id,JSON_EXTRACT(doc,'$.n')+0 n FROM src WHERE JSON_EXTRACT(doc,'$.n')+0 MEMBER OF(JSON_EXTRACT(doc,'$.tags')) OR NOT JSON_UNQUOTE(JSON_EXTRACT(doc,'$.s')) REGEXP '[0-9]+$' AND id=2 ORDER BY id;
SELECT o.id,d.score FROM (SELECT 1 id,2 v UNION ALL SELECT 2,5) o JOIN LATERAL (SELECT o.v*2+1 score) d ON d.score>o.id AND NOT d.score BETWEEN 4 AND 8 ORDER BY d.score DESC;
SELECT jt.n,jt.s FROM JSON_TABLE('[{"n":1,"s":"a1"},{"n":4,"s":"bbb"}]','$[*]' COLUMNS(n INT PATH '$.n',s VARCHAR(10) PATH '$.s')) jt WHERE NOT jt.n BETWEEN 2 AND 5 XOR jt.s REGEXP '[0-9]$' OR jt.n=4 AND jt.s LIKE 'b%' ORDER BY jt.n;
