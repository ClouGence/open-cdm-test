CREATE PROCEDURE split84.proc_cursor_diag(IN p_limit INT, OUT p_count INT, OUT p_state CHAR(5), OUT p_msg TEXT)
BEGIN
  DECLARE v_id INT DEFAULT 0;
  DECLARE v_data VARCHAR(20) DEFAULT '';
  DECLARE done BOOL DEFAULT FALSE;
  DECLARE cur1 CURSOR FOR SELECT id, data FROM split84.proc_src ORDER BY id;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
  DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 p_state = RETURNED_SQLSTATE, p_msg = MESSAGE_TEXT;
  END;
  SET p_count = 0;
  OPEN cur1;
  read_loop: LOOP
    FETCH NEXT FROM cur1 INTO v_id, v_data;
    IF done THEN
      LEAVE read_loop;
    END IF;
    SET p_count = p_count + 1;
    IF p_count >= p_limit THEN
      ITERATE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur1;
  GET DIAGNOSTICS @diag_count = NUMBER, @diag_rows = ROW_COUNT;
END;
CREATE PROCEDURE split84.proc_resignal_info()
BEGIN
  DECLARE bad CONDITION FOR SQLSTATE '45000';
  DECLARE EXIT HANDLER FOR bad
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @rs_errno = MYSQL_ERRNO, @rs_message = MESSAGE_TEXT;
    RESIGNAL SQLSTATE '45001' SET MESSAGE_TEXT = 'resignal changed', MYSQL_ERRNO = 1644;
  END;
  SIGNAL bad SET MESSAGE_TEXT = 'original signal', MYSQL_ERRNO = 1644;
END;




CREATE PROCEDURE proc_stacked_diag()
BEGIN
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @stack_state = RETURNED_SQLSTATE, @stack_message = MESSAGE_TEXT;
    RESIGNAL;
  END;
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'stacked diagnostics';
END;
