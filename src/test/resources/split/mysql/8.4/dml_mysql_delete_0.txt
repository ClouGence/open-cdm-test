DELETE d FROM split_dml_dst PARTITION(p0) AS d FORCE INDEX FOR JOIN (idx_v)

  WHERE EXISTS (SELECT 1 FROM split_dml_aux AS a WHERE a.id=c.id);

    WHERE JSON_CONTAINS_PATH(doc,'one','$.obsolete')
       OR EXISTS (
            SELECT 1
            FROM split_type_json.json_core AS inner_json
            WHERE inner_json.id<>json_core.id
              AND inner_json.doc->>'$.name'=json_core.doc->>'$.name'
          );
