WITH lob_cte(b,t) AS (
  SELECT b,t FROM lob_generated
  UNION ALL
  SELECT lb,lt FROM lob_generated
)
SELECT HEX(b),t FROM lob_cte WHERE b IS NOT NULL;
SELECT t,HEX(b),
       ROW_NUMBER() OVER (PARTITION BY t ORDER BY b) AS rn,
       FIRST_VALUE(t) OVER (ORDER BY b ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS first_t
FROM lob_generated;
