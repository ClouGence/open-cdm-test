WITH str_cte(c,v,b,vb) AS (
  SELECT c,v,b,vb FROM str_generated
  UNION ALL
  SELECT gc,gv,gb,gvb FROM str_generated
)
SELECT c,v,HEX(b),HEX(vb) FROM str_cte WHERE vb IS NOT NULL;
SELECT v, HEX(vb),
       ROW_NUMBER() OVER (PARTITION BY vb ORDER BY BINARY v) AS rn,
       FIRST_VALUE(c) OVER (ORDER BY b ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS first_c
FROM str_generated;
