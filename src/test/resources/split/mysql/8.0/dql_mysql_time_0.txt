SELECT id,t6,LAG(t6) OVER (ORDER BY t6) AS previous_time,MAX(t6) OVER (PARTITION BY HOUR(t6)) AS hour_max FROM time_widths;
WITH RECURSIVE time_cte(n,t) AS (SELECT 1,TIME'-00:00:00.000001' UNION ALL SELECT n+1,ADDTIME(t,'00:00:00.000001') FROM time_cte WHERE n<3) SELECT * FROM time_cte WHERE t IN (TIME'-00:00:00.000001',TIME'00:00:00.000001');
