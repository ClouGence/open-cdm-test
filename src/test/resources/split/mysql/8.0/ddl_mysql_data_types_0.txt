CREATE TABLE integer_generated (
  id INT PRIMARY KEY,
  tiny_value TINYINT,
  small_value SMALLINT,
  medium_value MEDIUMINT,
  int_value INTEGER,
  big_value BIGINT,
  signed_total BIGINT GENERATED ALWAYS AS (
    tiny_value+small_value+medium_value+int_value+big_value
  ) STORED,
  unsigned_mask BIGINT UNSIGNED GENERATED ALWAYS AS (
    CAST(int_value AS UNSIGNED)
  ) VIRTUAL,
  KEY idx_integer_generated (signed_total,unsigned_mask)
);








CREATE TABLE numeric_generated (
  id INT PRIMARY KEY,
  decimal_value DECIMAL(20,6),
  float_value FLOAT,
  double_value DOUBLE,
  exact_total DECIMAL(30,10) GENERATED ALWAYS AS (
    decimal_value+CAST(float_value AS DECIMAL(20,8))
  ) STORED,
  approximate_total DOUBLE GENERATED ALWAYS AS (
    float_value+double_value
  ) VIRTUAL,
  KEY idx_numeric_generated (exact_total,approximate_total)
);
CREATE TABLE temporal_generated_pair(dt DATETIME PRIMARY KEY,datetxt VARCHAR(10) GENERATED ALWAYS AS (DATE(dt)) STORED,timetxt VARCHAR(10) GENERATED ALWAYS AS (TIME(dt)) STORED) ENGINE=InnoDB;
CREATE TABLE generated_coercibility(a VARCHAR(1024),b INT GENERATED ALWAYS AS (COERCIBILITY(a)) VIRTUAL);
