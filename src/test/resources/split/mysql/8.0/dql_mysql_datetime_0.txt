SELECT id,CAST(ts AT TIME ZONE INTERVAL '-05:30' AS DATETIME(6)) AS shifted FROM temporal_modern;
SELECT dt FROM temporal_modern WHERE id<=2 INTERSECT SELECT CAST(ts AS DATETIME(6)) FROM temporal_modern WHERE id>=1;
SELECT dt FROM temporal_modern EXCEPT SELECT CAST(ts AS DATETIME(6)) FROM temporal_modern WHERE id=2;
SELECT id,ts,LAG(ts,1,TIMESTAMP'2000-01-01 00:00:00') OVER(ORDER BY dt) AS previous_ts,MAX(dt) OVER(ORDER BY ts RANGE BETWEEN INTERVAL 1 DAY PRECEDING AND CURRENT ROW) AS rolling_max FROM temporal_modern;
WITH RECURSIVE temporal_steps AS(SELECT TIMESTAMP'2020-01-01 00:00:00' AS dt UNION ALL SELECT dt+INTERVAL 1 DAY FROM temporal_steps WHERE dt<TIMESTAMP'2020-01-03 00:00:00') SELECT dt,TIMESTAMPDIFF(HOUR,TIMESTAMP'2020-01-01 00:00:00',dt) FROM temporal_steps;
SELECT DATE(ts),COUNT(*),MAX(dt),ROW_NUMBER() OVER(ORDER BY MAX(dt)) AS aggregate_position FROM temporal_modern GROUP BY DATE(ts) WITH ROLLUP HAVING COUNT(*)>0;
