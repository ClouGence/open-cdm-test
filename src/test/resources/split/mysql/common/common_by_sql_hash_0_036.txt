BEGIN
  DECLARE dbl DOUBLE;
  DECLARE deci DECIMAL(10, 2);
  DECLARE intv INTEGER;
  DECLARE str VARBINARY(16);
  SELECT DISTINCT VAR_POP(1) FROM proc_t1 GROUP BY pk INTO dbl;
  SELECT DISTINCT STDDEV_POP(1) FROM proc_t1 GROUP BY pk INTO dbl;
  SELECT DISTINCT STDDEV_SAMP(1) FROM proc_t1 GROUP BY pk INTO dbl;
  SELECT DISTINCT AVG(3.14159) FROM proc_t1 GROUP BY pk INTO deci;
  SELECT DISTINCT AVG(3.14159E0) FROM proc_t1 GROUP BY pk INTO dbl;
  SELECT DISTINCT BIT_OR(x'0000') FROM proc_t1 GROUP BY pk INTO str;
  SELECT DISTINCT BIT_OR(9) FROM proc_t1 GROUP BY pk INTO intv;
END;
