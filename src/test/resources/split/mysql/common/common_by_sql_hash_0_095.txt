CREATE PROCEDURE proc_compound_common(IN p_limit INT, OUT p_total INT)
COMMENT 'compound coverage'
LANGUAGE SQL
MODIFIES SQL DATA
SQL SECURITY INVOKER
BEGIN
  DECLARE v_id INT DEFAULT 0;
  DECLARE v_done BOOL DEFAULT FALSE;
  DECLARE cur_ids CURSOR FOR SELECT id FROM proc_src ORDER BY id;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
  SET p_total = 0;
  OPEN cur_ids;
  read_loop: LOOP
    FETCH NEXT FROM cur_ids INTO v_id;
    IF v_done THEN
      LEAVE read_loop;
    END IF;
    SET p_total = p_total + 1;
    CASE
      WHEN p_total < p_limit THEN SET v_id = v_id + 0;
      ELSE SET v_id = v_id;
    END CASE;
    IF p_total < p_limit THEN
      ITERATE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur_ids;
  WHILE p_total < p_limit DO
    SET p_total = p_total + 1;
  END WHILE;
  REPEAT
    SET p_total = p_total - 1;
  UNTIL p_total <= p_limit
  END REPEAT;
END;
