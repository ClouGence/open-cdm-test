CREATE DEFINER = CURRENT_USER EVENT IF NOT EXISTS split56.ev_once
  ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 1 HOUR + INTERVAL 1 MINUTE
  ON COMPLETION PRESERVE
  DISABLE
  COMMENT 'one time event'
  DO UPDATE split56.event_log SET note = 'once' WHERE id = 1;
CREATE EVENT split56.ev_repeat
  ON SCHEDULE EVERY 1 DAY
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 2 DAY
  ON COMPLETION NOT PRESERVE
  DISABLE ON SLAVE
  COMMENT 'repeat event'
  DO INSERT INTO split56.event_log VALUES (2, 'repeat');
ALTER EVENT split56.ev_once
  ON SCHEDULE EVERY 2 HOUR
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 1 DAY
  RENAME TO split56.ev_once_renamed
  ENABLE
  COMMENT 'renamed event'
  DO UPDATE split56.event_log SET note = 'renamed' WHERE id = 1;
ALTER EVENT split56.ev_repeat DISABLE ON SLAVE;
DROP EVENT IF EXISTS split56.ev_once_renamed;
DROP EVENT split56.ev_repeat;
