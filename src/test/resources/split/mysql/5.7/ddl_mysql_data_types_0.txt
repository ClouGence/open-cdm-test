CREATE TABLE audit_t (c JSON COLLATE utf8mb4_bin);






CREATE TABLE type_json (c_json JSON);





































CREATE TABLE bit_generated(a INT,b BIT(8) AS (a & 255) STORED,KEY idx_b(b));







CREATE TABLE time_generated(raw_value VARCHAR(32),generated_time TIME(6) GENERATED ALWAYS AS(CAST(raw_value AS TIME(6))) STORED);






CREATE TABLE temporal_generated(source_value VARCHAR(32),generated_datetime DATETIME(6) GENERATED ALWAYS AS(CAST(source_value AS DATETIME(6))) STORED,generated_timestamp TIMESTAMP(6) GENERATED ALWAYS AS(CAST(source_value AS DATETIME(6))) VIRTUAL,KEY idx_generated_datetime(generated_datetime),KEY idx_generated_timestamp(generated_timestamp));






























CREATE TABLE split_type_enum_set.es_generated (
      id INT PRIMARY KEY,
      e ENUM('low','medium','high'),
      s SET('read','write','admin'),
      e_index INT GENERATED ALWAYS AS (e+0) STORED,
      s_mask INT GENERATED ALWAYS AS (s+0) VIRTUAL,
      INDEX idx_e_index (e_index),
      INDEX idx_s_mask (s_mask)
    );
CREATE TABLE split_type_json.json_core (
      id INT PRIMARY KEY,
      doc JSON NOT NULL,
      payload JSON,
      doc_name VARCHAR(64)
        GENERATED ALWAYS AS (JSON_UNQUOTE(doc->'$.name')) STORED,
      KEY idx_doc_name (doc_name),
      CHECK (JSON_VALID(doc))
    );
CREATE TABLE split_type_json.json_primary_key (doc JSON PRIMARY KEY);
CREATE TABLE split_type_json.json_unique_key (doc JSON, UNIQUE KEY uq_doc(doc));
CREATE TABLE split_type_json.json_ordinary_key (doc JSON, KEY idx_doc(doc));
CREATE TABLE split_type_json.json_default_literal (id INT PRIMARY KEY,doc JSON DEFAULT '{}');



CREATE TABLE spatial_generated (
  id INT PRIMARY KEY,
  p POINT,
  g GEOMETRY AS (ST_Envelope(p)) STORED
);
ALTER TABLE spatial_generated
ADD COLUMN centroid POINT AS (ST_Centroid(g)) VIRTUAL;



CREATE TABLE integer_generated (
  id INT PRIMARY KEY,
  tiny_value TINYINT,
  small_value SMALLINT,
  medium_value MEDIUMINT,
  int_value INTEGER,
  big_value BIGINT,
  signed_total BIGINT GENERATED ALWAYS AS (
    tiny_value+small_value+medium_value+int_value+big_value
  ) STORED,
  unsigned_mask BIGINT UNSIGNED GENERATED ALWAYS AS (
    CAST(int_value AS UNSIGNED)
  ) VIRTUAL,
  KEY idx_integer_generated (signed_total,unsigned_mask)
);








CREATE TABLE numeric_generated (
  id INT PRIMARY KEY,
  decimal_value DECIMAL(20,6),
  float_value FLOAT,
  double_value DOUBLE,
  exact_total DECIMAL(30,10) GENERATED ALWAYS AS (
    decimal_value+CAST(float_value AS DECIMAL(20,8))
  ) STORED,
  approximate_total DOUBLE GENERATED ALWAYS AS (
    float_value+double_value
  ) VIRTUAL,
  KEY idx_numeric_generated (exact_total,approximate_total)
);
CREATE TABLE temporal_generated_pair(dt DATETIME PRIMARY KEY,datetxt VARCHAR(10) GENERATED ALWAYS AS (DATE(dt)) STORED,timetxt VARCHAR(10) GENERATED ALWAYS AS (TIME(dt)) STORED) ENGINE=InnoDB;
CREATE TABLE generated_coercibility(a VARCHAR(1024),b INT GENERATED ALWAYS AS (COERCIBILITY(a)) VIRTUAL);
