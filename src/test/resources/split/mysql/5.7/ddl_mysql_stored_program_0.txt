CREATE PROCEDURE proc_stacked_diag()
BEGIN
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @stack_state = RETURNED_SQLSTATE, @stack_message = MESSAGE_TEXT;
    RESIGNAL;
  END;
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'stacked diagnostics';
END;
DROP EVENT IF EXISTS split57.ev_once;
DROP EVENT IF EXISTS split57.ev_once_renamed;
DROP EVENT IF EXISTS split57.ev_repeat;
CREATE DEFINER = CURRENT_USER EVENT IF NOT EXISTS split57.ev_once
  ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 1 HOUR + INTERVAL 1 MINUTE
  ON COMPLETION PRESERVE
  DISABLE
  COMMENT 'one time event'
  DO UPDATE split57.event_log SET note = 'once' WHERE id = 1;
CREATE EVENT split57.ev_repeat
  ON SCHEDULE EVERY 1 DAY
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 2 DAY
  ON COMPLETION NOT PRESERVE
  DISABLE ON SLAVE
  COMMENT 'repeat event'
  DO INSERT INTO split57.event_log VALUES (2, 'repeat');
ALTER EVENT split57.ev_once
  ON SCHEDULE EVERY 2 HOUR
    STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
    ENDS CURRENT_TIMESTAMP + INTERVAL 1 DAY
  RENAME TO split57.ev_once_renamed
  ENABLE
  COMMENT 'renamed event'
  DO UPDATE split57.event_log SET note = 'renamed' WHERE id = 1;
ALTER EVENT split57.ev_repeat DISABLE ON SLAVE;
DROP EVENT IF EXISTS split57.ev_once_renamed;
DROP EVENT split57.ev_repeat;























CREATE PROCEDURE proc(ofs INT,count INT) BEGIN DECLARE i INT DEFAULT ofs; WHILE i<count DO IF LOWER(CHAR(i USING utf8))<>LOWER(CHAR(i USING utf8mb4)) THEN SELECT i AS 'found funny character'; END IF; SET i=i+1; END WHILE; END;

CREATE DEFINER = CURRENT_USER TRIGGER split57.trg_bi BEFORE INSERT ON split57.trigger_src FOR EACH ROW SET NEW.updated_at = CURRENT_TIMESTAMP;
CREATE TRIGGER split57.trg_bu BEFORE UPDATE ON split57.trigger_src FOR EACH ROW SET NEW.note = COALESCE(NEW.note, 'before update');
CREATE TRIGGER split57.trg_bd BEFORE DELETE ON split57.trigger_src FOR EACH ROW INSERT INTO split57.trigger_audit(src_id, action_name, old_amount, new_amount) VALUES (OLD.id, 'before delete', OLD.amount, NULL);
CREATE TRIGGER split57.trg_ai AFTER INSERT ON split57.trigger_src FOR EACH ROW INSERT INTO split57.trigger_audit(src_id, action_name, old_amount, new_amount) VALUES (NEW.id, 'after insert', NULL, NEW.amount);
CREATE TRIGGER split57.trg_au AFTER UPDATE ON split57.trigger_src FOR EACH ROW INSERT INTO split57.trigger_audit(src_id, action_name, old_amount, new_amount) VALUES (OLD.id, 'after update', OLD.amount, NEW.amount);
CREATE TRIGGER split57.trg_ad AFTER DELETE ON split57.trigger_src FOR EACH ROW INSERT INTO split57.trigger_audit(src_id, action_name, old_amount, new_amount) VALUES (OLD.id, 'after delete', OLD.amount, NULL);











CREATE PROCEDURE p11() BEGIN DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; SELECT 1; GET CURRENT DIAGNOSTICS @n=NUMBER; GET STACKED DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; END; DROP TABLE missing_t; END;
CREATE PROCEDURE p12() BEGIN DECLARE CONTINUE HANDLER FOR SQLWARNING BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; RESIGNAL SET MYSQL_ERRNO=9999,MESSAGE_TEXT='mapped'; GET CURRENT DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m4=MESSAGE_TEXT; END; SIGNAL SQLSTATE '01001'; END;
