CREATE TABLE aes_feedback_modes (a VARCHAR(128), b128 VARCHAR(144), b192 VARCHAR(144), b256 CHAR(144));

CREATE TABLE aes_ecb_modes (a VARCHAR(16), b128 CHAR(16), b192 CHAR(16), b256 CHAR(16));
CREATE TABLE analyse_ctas SELECT * FROM analyse_int PROCEDURE ANALYSE();
CREATE VIEW analyse_bad_view AS SELECT * FROM analyse_int PROCEDURE ANALYSE();
CREATE PROCEDURE analyse_select_proc()
BEGIN
  SELECT * FROM analyse_int PROCEDURE ANALYSE();
END;
CREATE PROCEDURE analyse_cursor_proc()
BEGIN
  DECLARE c1,c2,c3,c4,c5,c6,c7,c8,c9,c10 CHAR(20);
  DECLARE done INT DEFAULT 0;
  DECLARE cur1 CURSOR FOR SELECT * FROM analyse_int PROCEDURE ANALYSE();
  DECLARE CONTINUE HANDLER FOR SQLSTATE '02000' SET done=1;
  OPEN cur1;
  read_loop: LOOP
    FETCH cur1 INTO c1,c2,c3,c4,c5,c6,c7,c8,c9,c10;
    IF done THEN
      LEAVE read_loop;
    END IF;
  END LOOP;
  CLOSE cur1;
END;

















CREATE TABLE t_generated(x VARCHAR(10), gc INTEGER GENERATED ALWAYS AS (x LIKE 'abba' ESCAPE 'b'));




CREATE TABLE md5_generated (a VARCHAR(1024), b VARBINARY(32) GENERATED ALWAYS AS (MD5(a)) VIRTUAL);



CREATE FUNCTION split_r_like(input_text VARCHAR(255)) RETURNS BOOLEAN DETERMINISTIC RETURN REGEXP_LIKE(input_text, 'pattern');
CREATE FUNCTION split_r_instr(input_text VARCHAR(255)) RETURNS INT DETERMINISTIC RETURN REGEXP_INSTR(input_text, 'pattern');
CREATE FUNCTION split_r_replace(input_text VARCHAR(255)) RETURNS VARCHAR(255) DETERMINISTIC RETURN REGEXP_REPLACE(input_text, 'pattern', 'xyz');
CREATE FUNCTION split_r_substr(input_text VARCHAR(255)) RETURNS VARCHAR(255) DETERMINISTIC RETURN REGEXP_SUBSTR(input_text, 'pattern');

















CREATE TABLE t_gcol (
  a INT,
  b BLOB,
  c INT GENERATED ALWAYS AS (1) VIRTUAL,
  d INT,
  e INT GENERATED ALWAYS AS (LPAD('111', b, '1')) VIRTUAL,
  UNIQUE KEY (e),
  KEY(b(1),a,e),
  KEY(e,b(1))
);

CREATE TABLE split_func_system.ctas_charset SELECT CHARSET(_utf8'a'),COLLATION(_utf8'a');

















CREATE VIEW weight_level_view AS SELECT WEIGHT_STRING(value_text AS CHAR(2) LEVEL 1 DESC) AS weight_value FROM weight_source;
CREATE VIEW weight_level_list_view AS SELECT WEIGHT_STRING(value_text AS CHAR(2) LEVEL 1,3 REVERSE) AS weight_value FROM weight_source;
CREATE PROCEDURE split_limit57.proc_limit(IN p_offset INT,IN p_count INT)
BEGIN
  SELECT id FROM split_limit57.t ORDER BY id LIMIT p_count;
  SELECT id FROM split_limit57.t ORDER BY id LIMIT p_offset,p_count;
  SELECT id FROM split_limit57.t ORDER BY id LIMIT p_count OFFSET p_offset;
END;


CREATE TEMPORARY TABLE z2 (SELECT CAST(NULL AS TIME) AS c05 ORDER BY '1') ORDER BY '1';






























CREATE TABLE str_generated (
  c CHAR(8),
  v VARCHAR(16),
  b BINARY(8),
  vb VARBINARY(16),
  gc CHAR(8) GENERATED ALWAYS AS (c) VIRTUAL,
  gv VARCHAR(16) GENERATED ALWAYS AS (v) STORED,
  gb BINARY(8) GENERATED ALWAYS AS (b) VIRTUAL,
  gvb VARBINARY(16) GENERATED ALWAYS AS (vb) STORED
);




CREATE TABLE lob_generated (
  b BLOB,
  t TEXT,
  lb LONGBLOB,
  lt LONGTEXT,
  payload_length BIGINT GENERATED ALWAYS AS (OCTET_LENGTH(lb)+CHAR_LENGTH(lt)) STORED
);

CREATE VIEW split_type_json.json_view AS
    SELECT id,doc,doc->>'$.name' AS name_text,
           JSON_EXTRACT(payload,'$.tags') AS tags
    FROM split_type_json.json_core;
