SELECT id,doc,doc->'$.score',doc->>'$.name',
               JSON_TYPE(doc),JSON_DEPTH(doc),JSON_STORAGE_SIZE(doc)
        FROM split_type_json.json_core
        ORDER BY CAST(doc->>'$.score' AS UNSIGNED),id;
SELECT a.id,b.id,a.doc->>'$.name'
        FROM split_type_json.json_core AS a
        JOIN split_type_json.json_core AS b
          ON a.doc->>'$.kind'=b.doc->>'$.kind'
         AND JSON_OVERLAPS(a.payload,b.payload)
        WHERE a.id<b.id
        ORDER BY a.id,b.id;
SELECT o.id,o.doc
        FROM split_type_json.json_core AS o
        WHERE EXISTS (
          SELECT 1 FROM split_type_json.json_core AS i
          WHERE i.id<>o.id
            AND JSON_CONTAINS(
                  i.payload,JSON_QUOTE(o.doc->>'$.kind'),'$.tags'
                )
        );
SELECT d.kind,COUNT(*),JSON_ARRAYAGG(d.name_text),
               JSON_OBJECTAGG(d.id,d.score)
        FROM (
          SELECT id,doc->>'$.kind' AS kind,
                 doc->>'$.name' AS name_text,
                 CAST(doc->>'$.score' AS UNSIGNED) AS score
          FROM split_type_json.json_core
        ) AS d
        GROUP BY d.kind
        HAVING MAX(d.score)>=10;
SELECT doc FROM split_type_json.json_core WHERE id<3
        UNION ALL
        SELECT JSON_OBJECT(
                 'max_score',
                 (SELECT MAX(CAST(doc->>'$.score' AS UNSIGNED))
                  FROM split_type_json.json_core)
               );
