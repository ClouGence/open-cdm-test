WITH RECURSIVE graph(f, t, label) AS (
    VALUES (1, 2, 'arc 1 -> 2'),
           (1, 3, 'arc 1 -> 3'),
           (2, 4, 'arc 2 -> 4'),
           (4, 1, 'arc 4 -> 1')
), search_graph(f, t, label) AS (
    SELECT * FROM graph WHERE f = 1
    UNION DISTINCT
    SELECT g.* FROM graph g JOIN search_graph sg ON g.f = sg.t
) CYCLE f, t SET cycle_mark TO 'Y' DEFAULT 'N' USING path
SELECT f, t, label, cycle_mark FROM search_graph ORDER BY path LIMIT 8
