-- search+cycle
with recursive search_graph(f, t, label) as (
	select * from graph g
	union all
	select g.*
	from graph g, search_graph sg
	where g.f = sg.t
) search depth first by f, t set seq
  cycle f, t set is_cycle using path
select * from search_graph
;
select a, b, c
from (values (1, 2, 3), (4, null, 6), (7, 8, 9)) as t (a, b, c)
group by all rollup(a, b), rollup(a, c)
order by a, b, c
;
WITH RECURSIVE graph(f, t, label) AS (
    VALUES (1, 2, 'arc 1 -> 2'),
           (1, 3, 'arc 1 -> 3'),
           (2, 4, 'arc 2 -> 4'),
           (4, 1, 'arc 4 -> 1')
), search_graph(f, t, label) AS (
    SELECT * FROM graph WHERE f = 1
    UNION ALL
    SELECT g.* FROM graph g JOIN search_graph sg ON g.f = sg.t
) SEARCH BREADTH FIRST BY f, t SET seq
  CYCLE f, t SET is_cycle USING path
SELECT f, t, label, is_cycle FROM search_graph ORDER BY seq LIMIT 8
;
