CREATE FUNCTION "public"."explain_analyze_without_memory"("query" "text") RETURNS TABLE("out_line" "text")
    LANGUAGE "plpgsql"
    AS $$
declare
  line text;
begin
  for line in
    execute 'explain (analyze, costs off, summary off, timing off, buffers off) ' || query
  loop
    out_line := regexp_replace(line, '\d+kB', 'NNkB', 'g');
    return next;
  end loop;
end;
$$
;
