CREATE FUNCTION split_trigger_probe.audit_row()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, new_balance)
        VALUES (NEW.id, TG_ARGV[0], NEW.balance);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, old_balance, new_balance)
        VALUES (NEW.id, TG_ARGV[0], OLD.balance, NEW.balance);
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, old_balance)
        VALUES (OLD.id, TG_ARGV[0], OLD.balance);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$
