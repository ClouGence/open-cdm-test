SELECT c.relname, ns.nspname, listed_objects.column_list FROM pg_catalog.pg_class c
 JOIN pg_catalog.pg_namespace ns ON c.relnamespace OPERATOR(pg_catalog.=) ns.oid
 CROSS JOIN LATERAL (SELECT c.relkind IN ('p', 'I')) as p (inherited)
 LEFT JOIN pg_catalog.pg_class t ON c.reltoastrelid OPERATOR(pg_catalog.=) t.oid
 LEFT JOIN listed_objects ON listed_objects.object_oid OPERATOR(pg_catalog.=) c.oid
 WHERE c.relpersistence OPERATOR(pg_catalog.!=) 't'
 AND listed_objects.object_oid IS NOT NULL
 AND (
 EXISTS (SELECT NULL FROM pg_catalog.pg_attribute a
 WHERE a.attrelid OPERATOR(pg_catalog.=) c.oid
 AND a.attnum OPERATOR(pg_catalog.>) 0::pg_catalog.int2
 AND NOT a.attisdropped
 AND a.attstattarget IS DISTINCT FROM 0::pg_catalog.int2
 AND a.attgenerated OPERATOR(pg_catalog.<>) 'v'
 AND NOT EXISTS (SELECT NULL FROM pg_catalog.pg_statistic s
 WHERE s.starelid OPERATOR(pg_catalog.=) a.attrelid
 AND s.staattnum OPERATOR(pg_catalog.=) a.attnum
 AND s.stainherit OPERATOR(pg_catalog.=) p.inherited))
 OR EXISTS (SELECT NULL FROM pg_catalog.pg_statistic_ext e
 WHERE e.stxrelid OPERATOR(pg_catalog.=) c.oid
 AND e.stxstattarget IS DISTINCT FROM 0::pg_catalog.int2
 AND NOT EXISTS (SELECT NULL FROM pg_catalog.pg_statistic_ext_data d
 WHERE d.stxoid OPERATOR(pg_catalog.=) e.oid
 AND d.stxdinherit OPERATOR(pg_catalog.=) p.inherited))
 OR EXISTS (SELECT NULL FROM pg_catalog.pg_attribute a
 JOIN pg_catalog.pg_index i ON i.indexrelid OPERATOR(pg_catalog.=) a.attrelid
 WHERE i.indrelid OPERATOR(pg_catalog.=) c.oid
 AND i.indkey[a.attnum OPERATOR(pg_catalog.-) 1::pg_catalog.int2] OPERATOR(pg_catalog.=) 0::pg_catalog.int2
 AND a.attnum OPERATOR(pg_catalog.>) 0::pg_catalog.int2
 AND NOT a.attisdropped
 AND a.attstattarget IS DISTINCT FROM 0::pg_catalog.int2
 AND NOT EXISTS (SELECT NULL FROM pg_catalog.pg_statistic s
 WHERE s.starelid OPERATOR(pg_catalog.=) a.attrelid
 AND s.staattnum OPERATOR(pg_catalog.=) a.attnum
 AND s.stainherit OPERATOR(pg_catalog.=) p.inherited))
 OR EXISTS (SELECT NULL FROM pg_catalog.pg_attribute a
 WHERE a.attrelid OPERATOR(pg_catalog.=) c.oid
 AND a.attnum OPERATOR(pg_catalog.>) 0::pg_catalog.int2
 AND NOT a.attisdropped
 AND a.attstattarget IS DISTINCT FROM 0::pg_catalog.int2
 AND a.attgenerated OPERATOR(pg_catalog.<>) 'v'
 AND c.relhassubclass
 AND NOT p.inherited
 AND EXISTS (SELECT NULL FROM pg_catalog.pg_inherits h
 WHERE h.inhparent OPERATOR(pg_catalog.=) c.oid)
 AND NOT EXISTS (SELECT NULL FROM pg_catalog.pg_statistic s
 WHERE s.starelid OPERATOR(pg_catalog.=) a.attrelid
 AND s.staattnum OPERATOR(pg_catalog.=) a.attnum
 AND s.stainherit))

 )
