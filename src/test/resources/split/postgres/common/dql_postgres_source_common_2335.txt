SELECT
a.attrelid,
a.attnum,
a.attname,
a.attstattarget,
a.attstorage,
t.typstorage,
a.atthasdef,
a.attisdropped,
a.attlen,
a.attalign,
a.attislocal,
pg_catalog.format_type(t.oid, a.atttypmod) AS atttypname,
array_to_string(a.attoptions, ', ') AS attoptions,
CASE WHEN a.attcollation <> t.typcollation THEN a.attcollation ELSE 0 END AS attcollation,
pg_catalog.array_to_string(ARRAY(SELECT pg_catalog.quote_ident(option_name) || ' ' || pg_catalog.quote_literal(option_value) FROM pg_catalog.pg_options_to_table(attfdwoptions) ORDER BY option_name), E',
    ') AS attfdwoptions,
co.conname AS notnull_name,
CASE WHEN co.convalidated THEN pt.description ELSE NULL END AS notnull_comment,
CASE WHEN NOT co.convalidated THEN co.oid ELSE NULL END AS notnull_invalidoid,
co.connoinherit AS notnull_noinherit,
co.conislocal AS notnull_islocal,
a.attcompression AS attcompression,
a.attidentity,
CASE WHEN a.atthasmissing AND NOT a.attisdropped THEN a.attmissingval ELSE null END AS attmissingval,
a.attgenerated
FROM unnest('{16417,16424}'::pg_catalog.oid[]) AS src(tbloid)
JOIN pg_catalog.pg_attribute a ON (src.tbloid = a.attrelid) LEFT JOIN pg_catalog.pg_type t ON (a.atttypid = t.oid)
 LEFT JOIN pg_catalog.pg_constraint co ON (a.attrelid = co.conrelid
   AND co.contype = 'n' AND co.conkey = array[a.attnum])
 LEFT JOIN pg_catalog.pg_description pt ON (pt.classoid = co.tableoid AND pt.objoid = co.oid)
WHERE a.attnum > 0::pg_catalog.int2
ORDER BY a.attrelid, a.attnum
