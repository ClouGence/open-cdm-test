  pg_catalog.pg_get_function_arguments(p.oid) as "Argument data types",
 CASE p.prokind
  WHEN 'a' THEN 'agg'
  WHEN 'w' THEN 'window'
  WHEN 'p' THEN 'proc'
  ELSE 'func'

  WHEN p.provolatile = 'i' THEN 'immutable'

  WHEN p.proparallel = 'u' THEN 'unsafe'
