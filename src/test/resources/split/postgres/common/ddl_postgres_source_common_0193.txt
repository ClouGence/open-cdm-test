    AS $_$

  t text;
  l text;
BEGIN
  t = '';

           $q$
             EXPLAIN (TIMING off, COSTS off, VERBOSE on)
             SELECT * FROM oldtable ot FULL JOIN newtable nt USING (id)

  RAISE INFO '%', t;
  RETURN new;
