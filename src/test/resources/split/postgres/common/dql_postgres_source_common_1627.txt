       pg_catalog.pg_get_constraintdef(oid, true) AS condef
  FROM pg_catalog.pg_constraint c
 WHERE confrelid IN (SELECT pg_catalog.pg_partition_ancestors('22045')
                     UNION ALL VALUES ('22045'::pg_catalog.regclass))
       AND contype = 'f' AND conparentid = 0
ORDER BY conname
;
