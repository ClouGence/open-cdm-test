  VALUES ('"Foo"'::pg_catalog.regnamespace, NULL::pg_catalog.text),
  ('"Bar"'::pg_catalog.regnamespace, NULL::pg_catalog.text)
)
SELECT c.relname, ns.nspname, listed_objects.column_list FROM pg_catalog.pg_class c
 JOIN pg_catalog.pg_namespace ns ON c.relnamespace OPERATOR(pg_catalog.=) ns.oid
 LEFT JOIN pg_catalog.pg_class t ON c.reltoastrelid OPERATOR(pg_catalog.=) t.oid
 LEFT JOIN listed_objects ON listed_objects.object_oid OPERATOR(pg_catalog.=) ns.oid
 WHERE c.relpersistence OPERATOR(pg_catalog.!=) 't'
 AND listed_objects.object_oid IS NULL
 AND c.relkind OPERATOR(pg_catalog.=) ANY (array['r', 'm'])
 ORDER BY c.relpages DESC
;
