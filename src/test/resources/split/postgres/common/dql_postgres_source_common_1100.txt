exclude_pat (pattern_id, nsp_regex, rel_regex, heap_only, btree_only) AS (
SELECT pattern_id, nsp_regex, rel_regex, heap_only, btree_only FROM exclude_raw r
WHERE (r.db_regex IS NULL OR 'db2' ~ r.db_regex) AND (r.nsp_regex IS NOT NULL OR r.rel_regex IS NOT NULL)), relation (pattern_id, oid, nspname, relname, reltoastrelid, relpages, is_heap, is_btree) AS (
SELECT DISTINCT ON (c.oid) NULL::INTEGER AS pattern_id,
c.oid, n.nspname, c.relname, c.reltoastrelid, c.relpages, c.relam = 2 AS is_heap, c.relam = 403 AS is_btree
FROM pg_catalog.pg_class c INNER JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT OUTER JOIN exclude_pat ep
ON (n.nspname ~ ep.nsp_regex OR ep.nsp_regex IS NULL)
AND (c.relname ~ ep.rel_regex OR ep.rel_regex IS NULL)
AND (c.relam = 2 OR NOT ep.heap_only OR ep.rel_regex IS NULL)
AND (c.relam = 403 OR NOT ep.btree_only OR ep.rel_regex IS NULL)
WHERE c.relpersistence != 't'
AND ep.pattern_id IS NULL AND c.relam = 2 AND c.relkind IN ('r', 'm', 't') AND c.relnamespace != 99
ORDER BY c.oid), toast (oid, nspname, relname, relpages) AS (
SELECT t.oid, 'pg_toast', t.relname, t.relpages
FROM pg_catalog.pg_class t INNER JOIN relation r ON r.reltoastrelid = t.oid
LEFT OUTER JOIN exclude_pat ep
ON ('pg_toast' ~ ep.nsp_regex OR ep.nsp_regex IS NULL)
AND (t.relname ~ ep.rel_regex OR ep.rel_regex IS NULL)
AND ep.heap_only
WHERE ep.pattern_id IS NULL
AND t.relpersistence != 't'
), index (oid, nspname, relname, relpages) AS (
SELECT c.oid, r.nspname, c.relname, c.relpages FROM relation r
INNER JOIN pg_catalog.pg_index i ON r.oid = i.indrelid INNER JOIN pg_catalog.pg_class c ON i.indexrelid = c.oid AND c.relpersistence != 't'
INNER JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT OUTER JOIN exclude_pat ep ON (n.nspname ~ ep.nsp_regex OR ep.nsp_regex IS NULL) AND (c.relname ~ ep.rel_regex OR ep.rel_regex IS NULL) AND ep.btree_only
WHERE ep.pattern_id IS NULL AND c.relam = 403 AND c.relkind = 'i'
), toast_index (oid, nspname, relname, relpages) AS (
SELECT c.oid, 'pg_toast', c.relname, c.relpages FROM toast t INNER JOIN pg_catalog.pg_index i ON t.oid = i.indrelid
INNER JOIN pg_catalog.pg_class c ON i.indexrelid = c.oid AND c.relpersistence != 't'
WHERE true AND c.relam = 403 AND c.relkind = 'i')

FROM relation WHERE pattern_id IS NOT NULL UNION
