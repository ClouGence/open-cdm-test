WITH include_raw (pattern_id, rgx) AS (
SELECT NULL, NULL, NULL WHERE false),
exclude_raw (pattern_id, rgx) AS (
SELECT NULL, NULL, NULL WHERE false),
database (datname) AS (
SELECT d.datname FROM pg_catalog.pg_database d LEFT OUTER JOIN exclude_raw e ON d.datname ~ e.rgx 
WHERE d.datallowconn AND datconnlimit != -2 AND e.pattern_id IS NULL),
include_pat (pattern_id, checkable) AS (
SELECT i.pattern_id, COUNT(*) FILTER (WHERE d IS NOT NULL) AS checkable
FROM include_raw i LEFT OUTER JOIN database d ON d.datname ~ i.rgx
GROUP BY i.pattern_id),
filtered_databases (datname) AS (
SELECT DISTINCT d.datname FROM database d)
SELECT pattern_id, datname FROM (
SELECT pattern_id, NULL::TEXT AS datname FROM include_pat WHERE checkable = 0 UNION ALL
SELECT NULL, datname FROM filtered_databases) AS combined_records
ORDER BY pattern_id NULLS LAST, datname

WITH exclude_raw (pattern_id, db_regex, nsp_regex, rel_regex, heap_only, btree_only) AS (
VALUES
(0::INTEGER, NULL::TEXT, '^(public)$'::TEXT, NULL::TEXT, false::BOOLEAN, false::BOOLEAN),
(1::INTEGER, NULL::TEXT, '^(pg_catalog)$'::TEXT, NULL::TEXT, false::BOOLEAN, false::BOOLEAN),
(2::INTEGER, NULL::TEXT, '^(pg_toast)$'::TEXT, NULL::TEXT, false::BOOLEAN, false::BOOLEAN),
(3::INTEGER, NULL::TEXT, '^(information_schema)$'::TEXT, NULL::TEXT, false::BOOLEAN, false::BOOLEAN)
),
exclude_pat (pattern_id, nsp_regex, rel_regex, heap_only, btree_only) AS (
SELECT pattern_id, nsp_regex, rel_regex, heap_only, btree_only FROM exclude_raw r
WHERE (r.db_regex IS NULL OR 'postgres' ~ r.db_regex) AND (r.nsp_regex IS NOT NULL OR r.rel_regex IS NOT NULL)), relation (pattern_id, oid, nspname, relname, reltoastrelid, relpages, is_heap, is_btree) AS (
SELECT DISTINCT ON (c.oid) NULL::INTEGER AS pattern_id,
c.oid, n.nspname, c.relname, c.reltoastrelid, c.relpages, c.relam = 2 AS is_heap, c.relam = 403 AS is_btree
FROM pg_catalog.pg_class c INNER JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT OUTER JOIN exclude_pat ep
ON (n.nspname ~ ep.nsp_regex OR ep.nsp_regex IS NULL)
AND (c.relname ~ ep.rel_regex OR ep.rel_regex IS NULL)
AND (c.relam = 2 OR NOT ep.heap_only OR ep.rel_regex IS NULL)
AND (c.relam = 403 OR NOT ep.btree_only OR ep.rel_regex IS NULL)
WHERE c.relpersistence != 't'
AND ep.pattern_id IS NULL AND c.relam = 2 AND c.relkind IN ('r', 'm', 't') AND c.relnamespace != 99
ORDER BY c.oid), toast (oid, nspname, relname, relpages) AS (
SELECT t.oid, 'pg_toast', t.relname, t.relpages
FROM pg_catalog.pg_class t INNER JOIN relation r ON r.reltoastrelid = t.oid
LEFT OUTER JOIN exclude_pat ep
ON ('pg_toast' ~ ep.nsp_regex OR ep.nsp_regex IS NULL)
AND (t.relname ~ ep.rel_regex OR ep.rel_regex IS NULL)
AND ep.heap_only
WHERE ep.pattern_id IS NULL
AND t.relpersistence != 't'
), index (oid, nspname, relname, relpages) AS (
SELECT c.oid, r.nspname, c.relname, c.relpages FROM relation r
INNER JOIN pg_catalog.pg_index i ON r.oid = i.indrelid INNER JOIN pg_catalog.pg_class c ON i.indexrelid = c.oid AND c.relpersistence != 't'
INNER JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT OUTER JOIN exclude_pat ep ON (n.nspname ~ ep.nsp_regex OR ep.nsp_regex IS NULL) AND (c.relname ~ ep.rel_regex OR ep.rel_regex IS NULL) AND ep.btree_only
WHERE ep.pattern_id IS NULL AND c.relam = 403 AND c.relkind = 'i'
), toast_index (oid, nspname, relname, relpages) AS (
SELECT c.oid, 'pg_toast', c.relname, c.relpages FROM toast t INNER JOIN pg_catalog.pg_index i ON t.oid = i.indrelid
INNER JOIN pg_catalog.pg_class c ON i.indexrelid = c.oid AND c.relpersistence != 't'
WHERE true AND c.relam = 403 AND c.relkind = 'i')
SELECT pattern_id, is_heap, is_btree, oid, nspname, relname, relpages FROM (
SELECT pattern_id, is_heap, is_btree, NULL::OID AS oid, NULL::TEXT AS nspname, NULL::TEXT AS relname, NULL::INTEGER AS relpages
FROM relation WHERE pattern_id IS NOT NULL UNION
SELECT NULL::INTEGER AS pattern_id, is_heap, is_btree, oid, nspname, relname, relpages FROM relation UNION
SELECT NULL::INTEGER AS pattern_id, TRUE AS is_heap, FALSE AS is_btree, oid, nspname, relname, relpages FROM toast UNION
SELECT NULL::INTEGER AS pattern_id, FALSE AS is_heap, TRUE AS is_btree, oid, nspname, relname, relpages FROM index UNION
SELECT NULL::INTEGER AS pattern_id, FALSE AS is_heap, TRUE AS is_btree, oid, nspname, relname, relpages FROM toast_index
) AS combined_records ORDER BY relpages DESC NULLS FIRST, oid
;
