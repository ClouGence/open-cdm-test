-- Check insensitive cursor with INSERT
-- (The above tests don't test the SQL notion of an insensitive cursor
-- correctly, because per SQL standard, changes from WHERE CURRENT OF
-- commands should be visible in the cursor.  So here we make the
-- changes with a command that is independent of the cursor.)
BEGIN
;
SET ROLE regress_stats_superuser
;
rollback to s1_f
;
ALTER USER nonexistent SET application_name to 'BOMB'
;
DROP ROLE IF EXISTS split_reassign_b
;
GRANT SELECT,INSERT,REFERENCES,TRIGGER,TRUNCATE,MAINTAIN,UPDATE ON TABLE "public"."datdba_only" TO "pg_database_owner"
;
COPY pg_catalog.pg_default_acl (oid, defaclrole, defaclnamespace, defaclobjtype, defaclacl) TO stdout
;
EXECUTE dumpDomain('30946')
;
REINDEX SCHEMA public
;
set role to default
;
