    LANGUAGE "plpgsql"
    AS $$
declare
  node jsonb;
  group_stats jsonb;
  group_key text;
  space_key text;

  for node in select * from jsonb_array_elements(explain_analyze_inc_sort_nodes(query)) t loop
    for group_key in select unnest(array['Full-sort Groups', 'Pre-sorted Groups']::text[]) t loop

    end loop;
  end loop;

end;
