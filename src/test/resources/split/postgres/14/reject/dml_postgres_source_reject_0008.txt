MERGE INTO pa_target t
  USING (SELECT 1 as key, 'pa_merge2a' as val) s
  ON s.key = t.key
  WHEN NOT MATCHED THEN
	INSERT VALUES (s.key, s.val)
  WHEN MATCHED THEN
	UPDATE set key = t.key + 1, val = t.val || ' updated by ' || s.val
  WHEN NOT MATCHED BY SOURCE THEN
	UPDATE set key = t.key + 1, val = t.val || ' source not matched by pa_merge2a'
  RETURNING merge_action(), old, new, t.*
;

  WHEN NOT MATCHED THEN
    INSERT VALUES (slogts::timestamp, sid, delta, 'inserted by merge')
  RETURNING merge_action(), t.*

MERGE INTO ONLY split_merge18.inventory AS t
USING (VALUES ('keep', 3, 'fresh'), ('add', 7, 'new')) AS s(sku, delta, note)
ON t.sku = s.sku
WHEN MATCHED AND s.delta > 0 THEN UPDATE SET qty = t.qty + s.delta, note = s.note
WHEN NOT MATCHED BY TARGET THEN INSERT (sku, qty, note) VALUES (s.sku, s.delta, s.note)
WHEN NOT MATCHED BY SOURCE AND t.qty < 10 THEN DELETE
RETURNING WITH (OLD AS o, NEW AS n) merge_action() AS action, o.sku AS old_sku, n.sku AS new_sku, n.qty
;
