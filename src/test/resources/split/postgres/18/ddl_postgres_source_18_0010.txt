RETURNS float8
LANGUAGE SQL
IMMUTABLE STRICT
PARALLEL SAFE
RETURN CASE
  WHEN $1 < '0'::float8 THEN '0'::float8
  WHEN $1 / ('2'::float8 * earth()) > '1'::float8 THEN pi() * earth()
  ELSE '2'::float8 * earth() * asin($1 / ('2'::float8 * earth()))
END
