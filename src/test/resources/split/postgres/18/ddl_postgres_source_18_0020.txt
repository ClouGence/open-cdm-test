CREATE OR REPLACE FUNCTION latitude(earth)
RETURNS float8
LANGUAGE SQL
IMMUTABLE STRICT
PARALLEL SAFE
RETURN CASE
  WHEN public.cube_ll_coord($1, 3) / earth() < '-1'::float8 THEN '-90'::float8
  WHEN public.cube_ll_coord($1, 3) / earth() > '1'::float8 THEN '90'::float8
  ELSE degrees(asin(public.cube_ll_coord($1, 3) / earth()))
END
;
