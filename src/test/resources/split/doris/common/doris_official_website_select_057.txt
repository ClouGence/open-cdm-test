-- support array as object value
SELECT json_object('id', 1, 'level', array('"aaa"','"bbb"'));
-- support map as object value
SELECT json_object('id', 1, 'level', map('a', 'b', 'c', 'd'));
-- support struct as object value
SELECT json_object('id', 1, 'level', named_struct('name', 'a', 'age', 1));
-- support json as object value
SELECT json_object('id', 1, 'level', cast('{\"a\":\"b\"}' as JSON));

SELECT JSON_PARSE_NULLABLE_ERROR_TO_VALUE('{"name": "John", "age": }', 'default');
SELECT JSON_PARSE_NULLABLE_ERROR_TO_VALUE(NULL, 'default');
SELECT json_parse_error_to_invalid('{"k1":"v31","k2":300}');
SELECT json_parse_notnull('{"a":"b"}');
SELECT json_parse_error_to_invalid('invalid json');
SELECT json_parse_notnull('invalid json');
SELECT json_parse_error_to_value('invalid json', '{}');
SELECT json_quote('null'), json_quote('"null"');
SELECT json_quote('[1, 2, 3]');
SELECT json_quote(null);
select json_quote("\n\b\r\t");
select json_replace(null, null, null);
select json_replace('{"k": 1}', "$.k", 2);
select json_replace('{"k": 1}', "$.j", 2);
select json_replace(null, null, 's');
SELECT JSON_SEARCH('{"person": {"name": "John", "age": 30}, "name": "John"}', 'all', 'John');
SELECT JSON_SEARCH('{"person": {"name": "John", "age": 30}}', 'one', 'John', '$.person');
select json_set(null, null, null);
select json_set('{"k": 1}', "$.j", 2);
SELECT json_unquote('"doris"');

SELECT json_unquote('"\\ttest"');

select json_unquote('doris');
select json_unquote(1);
