  CAST(var["L_LINESTATUS"] AS TEXT),

  AVG(CAST(var["L_DISCOUNT"] AS DOUBLE))                                       AS AVG_DISC,
  COUNT(*)                                              AS COUNT_ORDER
FROM
  lineitem
WHERE
  CAST(var["L_SHIPDATE"] AS DATE) <= DATE '1998-12-01' - INTERVAL '90' DAY
GROUP BY
CAST(var["L_RETURNFLAG"] AS TEXT),
CAST(var["L_LINESTATUS"] AS TEXT)

CAST(var["L_RETURNFLAG"] AS TEXT),
CAST(var["L_LINESTATUS"] AS TEXT)
