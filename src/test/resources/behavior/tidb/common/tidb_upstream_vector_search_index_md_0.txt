sql:
-- For the following query, the `WHERE` filter is performed after KNN, so the vector index cannot be used:
SELECT * FROM
(
  SELECT * FROM vec_table
  ORDER BY VEC_COSINE_DISTANCE(embedding, '[1, 2, 3]')
  LIMIT 5
) t
WHERE category = "document";

/test/1/catalog1/schema1
expect:
{
  "SELECT": [
    {
      "subject": "Table(4:16~4:25) /test/1/catalog1/schema1/vec_table/",
      "action" : "READ"
    },
    {
      "subject": "Function(5:11~5:30) /test/1/catalog1/schema1/VEC_COSINE_DISTANCE/",
      "action" : "CALL"
    }
  ]
}
