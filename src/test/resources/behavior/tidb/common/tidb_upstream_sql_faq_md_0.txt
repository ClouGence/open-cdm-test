[select__common_tidb_upstream_sql_faq_md_0_000]
sql:
-- Returns 1 (TRUE)
-- utf8mb4_0900_ai_ci does not ignore trailing spaces
SELECT 'Apple ' = 'Apple' COLLATE utf8mb4_0900_ai_ci;
levels:
/test/1/catalog1/schema1
expect:
{
  "SELECT" : [ {
    "subject" : "Query(3:0~3:52) /test/1/",
    "action" : "READ"
  } ]
}
