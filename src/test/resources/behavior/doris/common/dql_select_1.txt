(
    SELECT
        s1t1.*,
        s2t2.column1 AS s2_col1,
        ROW_NUMBER() OVER (ORDER BY s1t2.column5) AS conflict_alias
    FROM schema1.table1 s1t1
    LEFT JOIN catalog2.schema2.table2 s2t2
      ON s1t1.column6 = s2t2.column3
    LEFT JOIN catalog2.schema1.table2 s1t2
      ON s1t1.column2 = s1t2.column4
)
UNION
(
    SELECT
        s2t1.*,
        s1t2.column2 AS s1_col2,
        RANK() OVER (PARTITION BY s2t1.column3 ORDER BY s2t1.column4) AS conflict_alias
    FROM schema2.table1 s2t1
    RIGHT JOIN catalog2.schema1.table2 s1t2
      ON s2t1.column5 = s1t2.column1
    WHERE s2t1.column6 IN (
        SELECT MAX(column6)
        FROM catalog2.schema2.table2
        GROUP BY column2
    )
)
ORDER BY
    CASE
        WHEN conflict_alias > 100 THEN column1
        ELSE column2
    END DESC;
